cmake_minimum_required(VERSION 3.24)
project("DeadWarrior 2" LANGUAGES CXX RC)
set(CAMPAIGN_SOURCES src/campaign_menu.cpp src/campaign_tutorial.cpp src/tutorial_defence.cpp
    src/tutorial_battle.cpp src/campaign_interaction.cpp src/campaign_archidon.cpp src/campaign_giants.cpp src/campaign_magic.cpp src/campaign_result.cpp src/campaign_progress.cpp src/campaign_intermission.cpp src/work_in_progress.cpp)
add_library(campaign_core STATIC ${CAMPAIGN_SOURCES})
target_compile_features(campaign_core PUBLIC cxx_std_20)
target_compile_options(campaign_core PRIVATE /W4 /permissive- /utf-8)
set_target_properties(campaign_core PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(DeadWarrior2 WIN32 src/windows.cpp src/windows_log.cpp src/windows_audio.cpp src/windows_display.cpp src/windows_render.cpp src/windows_save.cpp
    src/resources/windows.rc)
target_compile_features(DeadWarrior2 PRIVATE cxx_std_20)
target_compile_definitions(DeadWarrior2 PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(DeadWarrior2 PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/utf-8>")
target_include_directories(DeadWarrior2 PRIVATE src/resources)
target_link_libraries(DeadWarrior2 PRIVATE campaign_core d2d1 dwrite gdiplus winmm shell32 ole32 xaudio2)
set_target_properties(DeadWarrior2 PROPERTIES
    OUTPUT_NAME "DeadWarrior 2"
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
install(TARGETS DeadWarrior2 RUNTIME DESTINATION .)
install(DIRECTORY assets DESTINATION .)
add_executable(tutorial_probe EXCLUDE_FROM_ALL tools/check_tutorial_sim.cpp)
target_include_directories(tutorial_probe PRIVATE src)
target_compile_features(tutorial_probe PRIVATE cxx_std_20)
add_executable(archidon_probe EXCLUDE_FROM_ALL tools/check_archidon_sim.cpp)
target_include_directories(archidon_probe PRIVATE src)
target_compile_features(archidon_probe PRIVATE cxx_std_20)
add_executable(magic_probe EXCLUDE_FROM_ALL tools/check_magic_sim.cpp)
target_include_directories(magic_probe PRIVATE src)
target_compile_features(magic_probe PRIVATE cxx_std_20)
add_executable(rebels_probe EXCLUDE_FROM_ALL tools/check_rebels_sim.cpp)
target_include_directories(rebels_probe PRIVATE src)
target_compile_features(rebels_probe PRIVATE cxx_std_20)
add_executable(giants_probe EXCLUDE_FROM_ALL tools/check_giants_sim.cpp)
target_include_directories(giants_probe PRIVATE src)
target_compile_features(giants_probe PRIVATE cxx_std_20)
foreach(probe tutorial_probe archidon_probe magic_probe rebels_probe giants_probe)
    target_link_libraries(${probe} PRIVATE campaign_core)
    set_target_properties(${probe} PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endforeach()
add_executable(audio_probe EXCLUDE_FROM_ALL tools/check_audio.cpp src/windows_audio.cpp)
target_include_directories(audio_probe PRIVATE src)
target_compile_features(audio_probe PRIVATE cxx_std_20)
target_compile_definitions(audio_probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(audio_probe PRIVATE xaudio2 ole32 uuid)
