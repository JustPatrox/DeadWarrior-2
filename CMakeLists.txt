cmake_minimum_required(VERSION 3.24)
project("DeadWarrior 2" LANGUAGES C CXX)
set(CAMPAIGN_SOURCES src/campaign_menu.cpp src/raster_compositor.cpp src/campaign_tutorial.cpp src/campaign_toolbar.cpp src/tutorial_defence.cpp
    src/tutorial_battle.cpp src/campaign_interaction.cpp src/campaign_archidon.cpp src/campaign_giants.cpp src/campaign_bomber.cpp src/campaign_knight.cpp src/campaign_deads.cpp src/campaign_crawlers.cpp src/campaign_magic.cpp src/campaign_hill.cpp src/campaign_result.cpp src/campaign_progress.cpp src/campaign_intermission.cpp src/work_in_progress.cpp)
add_library(campaign_core STATIC ${CAMPAIGN_SOURCES})
target_compile_features(campaign_core PUBLIC cxx_std_20)
if(MSVC)
    target_compile_options(campaign_core PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(campaign_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_target_properties(campaign_core PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(DW2_SDL_FRONTEND "Use the portable SDL frontend (default outside Windows)" OFF)
if(NOT WIN32 OR DW2_SDL_FRONTEND)
    include(cmake/Portable.cmake)
else()
    enable_language(RC)
    add_executable(DeadWarrior2 WIN32 src/windows.cpp src/windows_log.cpp src/windows_audio.cpp src/windows_display.cpp src/windows_render.cpp src/windows_save.cpp src/pause_menu.cpp src/resources/windows.rc)
    target_compile_features(DeadWarrior2 PRIVATE cxx_std_20)
    target_compile_definitions(DeadWarrior2 PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(DeadWarrior2 PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/utf-8>")
    target_include_directories(DeadWarrior2 PRIVATE src/resources)
    target_link_libraries(DeadWarrior2 PRIVATE campaign_core d2d1 dwrite gdiplus winmm shell32 ole32 xaudio2)
    set_target_properties(DeadWarrior2 PROPERTIES OUTPUT_NAME "DeadWarrior 2" MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    install(TARGETS DeadWarrior2 RUNTIME DESTINATION .)
    install(DIRECTORY assets DESTINATION .)
endif()
